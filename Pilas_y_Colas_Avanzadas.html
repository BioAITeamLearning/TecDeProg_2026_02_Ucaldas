collections.deque: operaciones O(1) en ambos extremos#

deque es ideal para colas y deques (colas dobles) por su eficiencia en append, appendleft, pop, popleft.

from collections import deque

dq = deque([1,2,3])
dq.appendleft(0)

Pila (Stack) — Implementación desde cero (basada en lista)#

Interfaz mínima: push, pop, peek, is_empty, size.

class Stack:
    def __init__(self):
        """Inicializa una pila vacía"""
        self.items = []

    def push(self, item):
        """Añade un elemento al tope"""
        self.items.append(item)

    def pop(self):
        """Elimina y devuelve el elemento del tope. Devuelve None si está vacía."""
        if self.is_empty():
            return None
        return self.items.pop()

    def peek(self):
        """Devuelve el elemento del tope sin eliminarlo. None si está vacía."""
        if self.is_empty():
            return None
        return self.items[-1]

    def is_empty(self):
        return len(self.items) == 0

    def size(self):
        return len(self.items)

# Prueba rápida
s = Stack()
for x in "ABC":
    s.push(x)
(s.size(), s.peek(), s.pop(), s.pop(), s.pop(), s.pop())

(3, 'C', 'C', 'B', 'A', None)

Ejemplo con pila: Conversión decimal → binario#

def decimal_a_binario(numero:int) -> str:
    if numero == 0:
        return "0"
    pila = Stack()
    n = numero
    while n > 0:
        pila.push(str(n % 2))
        n //= 2
    # Revertimos con pop (LIFO)
    bits = []
    while not pila.is_empty():
        bits.append(pila.pop())
    return "".join(bits)

for n in [0,1,2,5,23,42,255]:
    print(n, "->", decimal_a_binario(n))

1 -> 1
2 -> 10
5 -> 101
23 -> 10111
42 -> 101010
255 -> 11111111

Ejemplo con pila: Balanceo de paréntesis/llaves/corchetes#

def verificar_parentesis(expresion:str) -> bool:
    pila = Stack()
    pares = {'(' : ')', '[' : ']', '{' : '}'}
    cierres = set(pares.values())
    for ch in expresion:
        if ch in pares:
            pila.push(ch)
        elif ch in cierres:
            if pila.is_empty():
                return False
            apertura = pila.pop()
            if pares[apertura] != ch:
                return False
    return pila.is_empty()

tests = ["()[]{}", "([{}])", "([)]", "(", "", "([]{})", "([]{)"]
{t: verificar_parentesis(t) for t in tests}

 '([{}])': True,
 '([)]': False,
 '(': False,
 '': True,
 '([]{})': True,
 '([]{)': False}

Pila avanzada: MinStack (mínimo en O(1))#

Mantiene una pila auxiliar con los mínimos.

class MinStack:
    def __init__(self):
        self.stack = []
        self.mins = []
    def push(self, x):
        self.stack.append(x)
        if not self.mins or x <= self.mins[-1]:
            self.mins.append(x)
    def pop(self):
        if not self.stack:
            return None
        x = self.stack.pop()
        if x == self.mins[-1]:
            self.mins.pop()
        return x
    def top(self):
        return self.stack[-1] if self.stack else None
    def get_min(self):
        return self.mins[-1] if self.mins else None

m = MinStack()
for v in [5,3,7,2,2,8]:
    m.push(v)
(min_before := m.get_min(), m.pop(), m.get_min())

(2, 8, 2)

Cola (Queue): Implementación ineficiente con lista vs eficiente con deque#

Evita pop(0) por ser O(n). Usa deque para O(1) en popleft().

class QueueIneficiente:
    def __init__(self):
        self.items = []
    def enqueue(self, x):
        self.items.append(x)      # O(1)
    def dequeue(self):
        if self.items:
            return self.items.pop(0)  # O(n) — desplaza todos
        return None
    def is_empty(self):
        return len(self.items) == 0

from collections import deque

class Queue:
    def __init__(self):
        self.items = deque()
    def enqueue(self, x):
        self.items.append(x)      # O(1)
    def dequeue(self):
        if self.items:
            return self.items.popleft()  # O(1)
        return None
    def front(self):
        return self.items[0] if self.items else None
    def rear(self):
        return self.items[-1] if self.items else None
    def is_empty(self):
        return len(self.items) == 0

Ejemplo: Simulación simple de turnos en banco (sin sleep)#

import random

def simular_banco(nombres, seed=0):
    random.seed(seed)
    cola = Queue()
    for nombre in nombres:
        cola.enqueue(nombre)
    tiempo_total = 0
    atendidos = []
    while not cola.is_empty():
        cliente = cola.dequeue()
        t = random.randint(2,5)  # minutos
        tiempo_total += t
        atendidos.append((cliente, t))
    return atendidos, tiempo_total

atendidos, total = simular_banco(["Ana","Luis","María","Carlos","Elena"])
atendidos, total

Caso avanzado: Planificación Round-Robin#

class Proceso:
    def __init__(self, nombre, tiempo_cpu):
        self.nombre = nombre
        self.tiempo_restante = int(tiempo_cpu)
        self.tiempo_original = int(tiempo_cpu)
    def __repr__(self):
        return f"{self.nombre}(restante={self.tiempo_restante})"

def round_robin(procesos, quantum=3):
    cola = Queue()
    for p in procesos:
        cola.enqueue(p)
    timeline = []
    tiempo = 0
    while not cola.is_empty():
        p = cola.dequeue()
        ejec = min(quantum, p.tiempo_restante)
        p.tiempo_restante -= ejec
        timeline.append((tiempo, p.nombre, ejec))
        tiempo += ejec
        if p.tiempo_restante > 0:
            cola.enqueue(p)
    return timeline, tiempo

procs = [Proceso("P1",5), Proceso("P2",9), Proceso("P3",4)]
tl, total = round_robin(procs, quantum=3)
tl, total

Cola de prioridad (heapq): menor número = mayor prioridad#

import heapq

class ColaPrioridad:
    def __init__(self):
        self.heap = []
    def enqueue(self, item, prioridad:int):
        heapq.heappush(self.heap, (prioridad, item))
    def dequeue(self):
        if not self.heap:
            return None
        return heapq.heappop(self.heap)[1]
    def is_empty(self):
        return len(self.heap) == 0

hospital = ColaPrioridad()
hospital.enqueue("Dolor de cabeza", 5)
hospital.enqueue("Infarto", 1)
hospital.enqueue("Fractura", 3)
[hospital.dequeue(), hospital.dequeue(), hospital.dequeue()]

['Infarto', 'Fractura', 'Dolor de cabeza']

Deque (cola doble) personalizada: insertar/eliminar por ambos lados#

from collections import deque

class ColaDoble:
    def __init__(self):
        self.items = deque()
    # Frente
    def add_front(self, x):
        self.items.appendleft(x)
    def remove_front(self):
        return self.items.popleft() if self.items else None
    # Parte trasera
    def add_rear(self, x):
        self.items.append(x)
    def remove_rear(self):
        return self.items.pop() if self.items else None
    def is_empty(self):
        return len(self.items) == 0

nav = ColaDoble()
nav.add_rear("google.com")
nav.add_rear("github.com")
nav.add_front("stackoverflow.com")
list(nav.items)

['stackoverflow.com', 'google.com', 'github.com']

Estructuras circulares: Pila/Cola con capacidad fija#

Cola circular con punteros front y rear y Pila circular que sobrescribe lo más antiguo.

class CircularQueue:
    def __init__(self, capacidad:int):
        assert capacidad > 0
        self.cap = capacidad
        self.data = [None]*capacidad
        self.size = 0
        self.front = 0  # índice del próximo a salir
        self.rear = -1  # índice del último insertado
    def enqueue(self, x):
        if self.size == self.cap:
            # sobrescribe el más antiguo (avanza front también)
            self.front = (self.front + 1) % self.cap
            self.rear = (self.rear + 1) % self.cap
            self.data[self.rear] = x
        else:
            self.rear = (self.rear + 1) % self.cap
            self.data[self.rear] = x
            self.size += 1
    def dequeue(self):
        if self.size == 0:
            return None
        x = self.data[self.front]
        self.data[self.front] = None
        self.front = (self.front + 1) % self.cap
        self.size -= 1
        return x
    def __repr__(self):
        return f"CQ(data={self.data}, front={self.front}, rear={self.rear}, size={self.size})"

cq = CircularQueue(3)
for x in [1,2,3,4,5]:
    cq.enqueue(x)
repr(cq), cq.dequeue(), repr(cq)

 'CQ(data=[4, 5, None], front=0, rear=1, size=2)')

class PilaCircular:
    def __init__(self, capacidad:int):
        assert capacidad > 0
        self.capacidad = capacidad
        self.items = [None]*capacidad
        self.tope = -1
        self.tamaño = 0
    def push(self, item):
        # sobrescribe el más antiguo al llenarse
        self.tope = (self.tope + 1) % self.capacidad
        self.items[self.tope] = item
        if self.tamaño < self.capacidad:
            self.tamaño += 1
    def pop(self):
        if self.tamaño == 0:
            return None
        x = self.items[self.tope]
        self.items[self.tope] = None
        self.tope = (self.tope - 1) % self.capacidad
        self.tamaño -= 1
        return x
    def is_full(self):
        return self.tamaño == self.capacidad
    def is_empty(self):
        return self.tamaño == 0

pc = PilaCircular(3)
for v in [10,20,30,40]:
    pc.push(v)
(pc.items, pc.pop(), pc.items)

([None, 20, 30], 40, [None, 20, 30])

Recorrido de grafos: BFS con cola#

from collections import defaultdict

def bfs(grafo, inicio):
    visit = set([inicio])
    cola = Queue()
    cola.enqueue(inicio)
    orden = []
    while not cola.is_empty():
        u = cola.dequeue()
        orden.append(u)
        for v in grafo[u]:
            if v not in visit:
                visit.add(v)
                cola.enqueue(v)
    return orden

grafo = defaultdict(list, {
    'A': ['B','C'],
    'B': ['D','E'],
    'C': ['F'],
    'D': [], 'E': ['F'], 'F': []
})
bfs(grafo, 'A')

['A', 'B', 'C', 'D', 'E', 'F']

Backtracking con pila implícita (recursión) o explícita#

Ejemplo: Generar todas las combinaciones balanceadas de paréntesis para n pares.

def generar_parentesis(n:int):
    res = []
    def bt(cad, abiertos, cerrados):
        if len(cad) == 2*n:
            res.append(cad); return
        if abiertos < n:
            bt(cad+'(', abiertos+1, cerrados)
        if cerrados < abiertos:
            bt(cad+')', abiertos, cerrados+1)
    bt("", 0, 0)
    return res

generar_parentesis(3)

['((()))', '(()())', '(())()', '()(())', '()()()']

Micro-benchmark: lista vs deque (inserción/eliminación al inicio)#

Nota: Micro-benchmarks varían por máquina; el objetivo es observar la tendencia.

import time
from collections import deque
import matplotlib.pyplot as plt

def bench(n=50_000):
    # push-front y pop-front con list vs deque
    t0 = time.time()
    L = []
    for i in range(n):
        L.insert(0, i)
    for i in range(n):
        L.pop(0)
    t_list = time.time()-t0

    t0 = time.time()
    D = deque()
    for i in range(n):
        D.appendleft(i)
    for i in range(n):
        D.popleft()
    t_deque = time.time()-t0
    return t_list, t_deque

t_list, t_deque = bench(n=20_000)
print("list (insert(0)/pop(0)):", round(t_list,4), "s")
print("deque (appendleft/popleft):", round(t_deque,4), "s")

# Gráfico simple
plt.figure()
plt.bar(["list inicio","deque inicio"], [t_list, t_deque])
plt.title("Comparación: list vs deque (operaciones al inicio)")
plt.ylabel("Tiempo (s)")
plt.show()

list (insert(0)/pop(0)): 0.0674 s
deque (appendleft/popleft): 0.001 s

Ejercicio práctico: Sistema de Turnos Médicos (con prioridad)#

Requerimiento: gestionar cola general y cola de urgencias (prioridad). Atender siempre primero urgencias.

from collections import deque

class SistemaTurnos:
    def __init__(self):
        self.cola_general = deque()
        self.cola_urgencia = deque()
        self.pacientes_atendidos = 0

    def agregar_paciente(self, nombre, es_urgencia=False):
        if es_urgencia:
            self.cola_urgencia.append(nombre)
        else:
            self.cola_general.append(nombre)

    def atender(self):
        """Atiende al siguiente paciente (urgencia primero). Devuelve el nombre o None."""
        if self.cola_urgencia:
            self.pacientes_atendidos += 1
            return self.cola_urgencia.popleft()
        if self.cola_general:
            self.pacientes_atendidos += 1
            return self.cola_general.popleft()
        return None

    def pendientes(self):
        return len(self.cola_urgencia) + len(self.cola_general)

# Demo
sis = SistemaTurnos()
sis.agregar_paciente("Ana")
sis.agregar_paciente("Luis", es_urgencia=True)
sis.agregar_paciente("María")
sis.agregar_paciente("Carlos", es_urgencia=True)
orden = []
while sis.pendientes():
    orden.append(sis.atender())
orden, sis.pacientes_atendidos

(['Luis', 'Carlos', 'Ana', 'María'], 4)
