Resumen de la Unidad · Pila, Cola y Cola Circular#

Hasta aquí viste tres estructuras y las escribiste con funciones sueltas, y después aprendiste a escribir clases. Este cuaderno junta las dos cosas.

Tiene dos partes:

  1. Un resumen de referencia: todos los métodos de cada estructura, en tablas, para consultar.

  2. Un encargo: implementar las tres como clases, con un banco de pruebas que te dice si te quedaron bien.

Important

Las tres clases que construyas aquí las vamos a usar en el resto del curso. En la Unidad 2 la pila reaparece en recursión y backtracking; en la Unidad 3, las colas en los recorridos. No es un ejercicio desechable: es tu caja de herramientas.

3. El encargo#

Implementa las tres clases: Pila, Cola y ColaCircular.

Requisitos comunes a las tres:

  1. La colección interna se llama _datos (guion bajo: es interna).

  2. Se puede construir vacía o desde un iterable: Pila([1, 2, 3]) deja el 3 en la cima; Cola([1, 2, 3]) deja el 1 al frente. (ColaCircular recibe la capacidad, no un iterable.)

  3. Los errores se lanzan, no se devuelven: nada de retornar None cuando está vacía. Usa IndexError, OverflowError y ValueError según la tabla.

  4. Implementa los protocolos: __len__, __bool__, __iter__, __repr__ y, salvo en la circular, __contains__.

  5. cima, frente, final y capacidad son propiedades, no métodos: se usan sin paréntesis.

  6. Cola debe usar collections.deque por dentro, no una lista: sacar del frente de una lista obliga a correr de lugar todos los demás elementos, y deque no.

Debajo de cada clase hay un banco de pruebas. Ejecútalo: te dice exactamente qué falta.

Banco de pruebas#

Ejecuta esta celda una vez antes de empezar. No la modifiques.

# ============ BANCO DE PRUEBAS (identico al del cuaderno) ============
def _probar(nombre, casos):
    ok = 0
    for etiqueta, fn in casos:
        try:
            fn(); print(f"  OK     {etiqueta}"); ok += 1
        except AssertionError as e:
            print(f"  FALLA  {etiqueta}  -> {e or 'el resultado no es el esperado'}")
        except Exception as e:
            print(f"  ERROR  {etiqueta}  -> {type(e).__name__}: {e}")
    print(f"\n  {nombre}: {ok}/{len(casos)} pruebas superadas\n")

def _lanza(excepcion, fn):
    try: fn()
    except excepcion: return
    except Exception as e: raise AssertionError(f"lanzo {type(e).__name__} en vez de {excepcion.__name__}")
    raise AssertionError(f"no lanzo {excepcion.__name__}")

def probar_pila(Pila):
    def construccion():
        assert len(Pila()) == 0
        assert len(Pila([1, 2, 3])) == 3
        assert Pila([1, 2, 3]).cima == 3, "el ultimo del iterable debe quedar en la cima"
    def lifo():
        p = Pila(); p.apilar("A"); p.apilar("B"); p.apilar("C")
        assert p.desapilar() == "C" and p.desapilar() == "B" and p.desapilar() == "A"
    def cima_no_saca():
        p = Pila([1, 2]); assert p.cima == 2 and len(p) == 2, "cima no debe eliminar"
    def vacias():
        p = Pila()
        assert p.esta_vacia() and not p and len(p) == 0
        _lanza(IndexError, p.desapilar)
        _lanza(IndexError, lambda: p.cima)
    def protocolos():
        p = Pila([1, 2, 3])
        assert bool(p) is True and 2 in p and 9 not in p
        assert list(p) == [3, 2, 1], "__iter__ debe ir de la cima hacia la base"
        assert "Pila" in repr(p), "__repr__ debe nombrar la clase"
    def encapsulado():
        p = Pila([1, 2])
        assert not hasattr(p, "datos"), "la lista interna debe llamarse _datos, no datos"
    _probar("Pila", [("construccion", construccion), ("orden LIFO", lifo),
                     ("cima no elimina", cima_no_saca), ("pila vacia", vacias),
                     ("protocolos", protocolos), ("encapsulamiento", encapsulado)])

def probar_cola(Cola):
    def construccion():
        assert len(Cola()) == 0 and len(Cola([1, 2, 3])) == 3
        assert Cola([1, 2, 3]).frente == 1, "el primero del iterable debe quedar al frente"
    def fifo():
        c = Cola(); c.encolar("A"); c.encolar("B"); c.encolar("C")
        assert c.desencolar() == "A" and c.desencolar() == "B" and c.desencolar() == "C"
    def extremos():
        c = Cola([1, 2, 3])
        assert c.frente == 1 and c.final == 3 and len(c) == 3, "frente/final no deben eliminar"
    def vacias():
        c = Cola()
        assert c.esta_vacia() and not c
        _lanza(IndexError, c.desencolar)
        _lanza(IndexError, lambda: c.frente)
        _lanza(IndexError, lambda: c.final)
    def protocolos():
        c = Cola([1, 2, 3])
        assert 2 in c and 9 not in c
        assert list(c) == [1, 2, 3], "__iter__ debe ir del frente hacia el final"
        assert "Cola" in repr(c)
    _probar("Cola", [("construccion", construccion), ("orden FIFO", fifo),
                     ("frente y final", extremos), ("cola vacia", vacias),
                     ("protocolos", protocolos)])

def probar_cola_circular(ColaCircular):
    def construccion():
        c = ColaCircular(3)
        assert c.capacidad == 3 and len(c) == 0 and c.esta_vacia()
        _lanza(ValueError, lambda: ColaCircular(0))
    def llenado():
        c = ColaCircular(3)
        for x in "ABC": c.encolar(x)
        assert c.esta_llena() and len(c) == 3
        _lanza(OverflowError, lambda: c.encolar("D"))
    def da_la_vuelta():
        c = ColaCircular(4)
        for x in "ABCD": c.encolar(x)
        assert c.desencolar() == "A" and c.desencolar() == "B"
        c.encolar("E"); c.encolar("F")
        assert list(c) == ["C", "D", "E", "F"], f"orden logico incorrecto: {list(c)}"
        assert len(c) == 4 and c.esta_llena()
    def capacidad_fija():
        c = ColaCircular(2)
        for _ in range(20):
            c.encolar(1); c.desencolar()
        assert c.capacidad == 2, "la capacidad no puede cambiar"
        assert len(c._datos) == 2, "el arreglo interno no debe crecer"
    def vacias():
        c = ColaCircular(2)
        _lanza(IndexError, c.desencolar)
        _lanza(IndexError, lambda: c.frente)
    def solo_lectura():
        c = ColaCircular(3)
        _lanza(AttributeError, lambda: setattr(c, "capacidad", 99))
    _probar("ColaCircular", [("construccion", construccion), ("se llena y rechaza", llenado),
                             ("da la vuelta", da_la_vuelta), ("capacidad fija", capacidad_fija),
                             ("cola vacia", vacias), ("capacidad de solo lectura", solo_lectura)])

7. Ejercicios aplicados#

Con las tres clases funcionando, resuelve estos problemas usándolas. No vuelvas a list ni a deque directamente: si tu caja de herramientas sirve, esto debe salir con ella.

  1. Paréntesis balanceados (Pila) — esta_balanceado(texto) que verifique que cada (, [ y { se cierra con su pareja y en el orden correcto. "({[]})" → True, "(]" → False.

  2. Deshacer y rehacer (dos Pilas) — un editor con escribir(texto), deshacer() y rehacer(). Al escribir algo nuevo después de deshacer, la pila de rehacer se vacía.

  3. Invertir una cola (Pila + Cola) — invertir(cola) que le dé la vuelta usando solo una pila auxiliar. Sin reversed, sin rebanadas.

  4. Atención de turnos (Cola) — simula una ventanilla: llegan personas, se atiende una por vuelta, y en cada paso imprime a quién se atendió y cuántos quedan esperando.

  5. Las últimas N lecturas (ColaCircular) — registrar(lectura) que guarde siempre las últimas 5, pisando la más vieja cuando se llena. (Pista: la clase lanza OverflowError; captúralo y desencola antes de reintentar.)

  6. Round-Robin (ColaCircular) — tres procesos necesitan 3, 1 y 2 turnos de CPU. Saca el del frente, réstale un turno y, si le quedan, vuélvelo a encolar. Imprime el orden de atención.

  7. Para pensar (sin código) — el ejercicio 3 usa una pila para invertir una cola. ¿Podrías invertir una pila usando solo una cola auxiliar? ¿Por qué?
